### 1. 手动评测

逐条输入测试用例，记录实际结果：

```markdown

|---------|---------|---------|---------|---------|---------|
| BUG-001 | bug     | bug     | 创建Issue | 创建Issue | ✅ |
| ...     | ...     | ...     | ...     | ...     | ... |
```

### 2. 自动化评测

使用 E2E 测试框架批量执行：

```typescript
// tests/e2e/golden-eval.spec.ts
import { test, expect } from '@playwright/test';
import { goldenTestCases } from './golden-eval-data';

test.describe('Golden Evaluation Tests', () => {
  for (const testCase of goldenTestCases) {
    test(`${testCase.id}: ${testCase.expectedType}`, async ({ page }) => {
      await page.goto('/');
      await page.getByLabel('反馈内容').fill(testCase.input);
      await page.getByRole('button', { name: '提交反馈' }).click();
      
      const alert = page.locator('[role="alert"]').last();
      await expect(alert).toBeVisible({ timeout: 60000 });
      
      const alertText = await alert.textContent();
      
      if (testCase.shouldPass) {
        expect(alertText).toMatch(/已创建|成功/i);
      } else {
        expect(alertText).toMatch(/拒绝|无法|错误/i);
      }
    });
  }
});
```

```typescript
// 分类准确率
```
